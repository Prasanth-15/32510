fetchWithTimeout(resource, options = {}, timeout = 8000) {
      const controller = new AbortController();
      options.signal = controller.signal;
      const id = setTimeout(() => controller.abort(), timeout);
      try {
        const res = await fetch(resource, options);
        clearTimeout(id);
        return res;
      } catch (err) {
        clearTimeout(id);
        throw err;
      }
    }

function validateAndSanitizeUser(payload) {
  if (!payload || typeof payload !== "object") return { ok: false, error: "Invalid payload" };
  let name = String(payload.name ?? "").trim();
  let role = String(payload.role ?? "User").trim();
  if (!name) return { ok: false, error: "Name is required" };
  if (name.length > 100) name = name.slice(0, 100);
  const allowedRoles = new Set(["User", "Admin"]);
  if (!allowedRoles.has(role)) role = "User";
  name = name.replace(/[\x00-\x1F\x7F]/g, "");
  return { ok: true, user: { name, role } };
}

// --- API routes ---
app.get("/api/users", (req, res) => {
  db.all("SELECT id, name, role, created_at FROM users ORDER BY id DESC LIMIT 1000", (err, rows) => {
    if (err) return res.status(500).json({ error: "Database error" });
    res.json(rows || []);
  });
});

app.post("/api/users", (req, res) => {
  const result = validateAndSanitizeUser(req.body);
  if (!result.ok) return res.status(400).json({ error: result.error });
  const { name, role } = result.user;
  const created_at = new Date().toISOString();
  db.run("INSERT INTO users (name, role, created_at) VALUES (?, ?, ?)", [name, role, created_at], function (err) {
